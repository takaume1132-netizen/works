毎月くり返している表の作業を、ボタン1つで終わらせます

ご覧いただきありがとうございます。Google スプレッドシートや Excel で毎回くり返している転記・集計・書類作成・メール送信などを自動化しています。よくある作業を5つ見本にしましたので、このページで実際にお試しください。

次のような作業はありませんか

同じ内容を複数の表に転記している

現在
予定表・台帳・担当者へのメールの3か所に、同じ内容を入力している。入力ミスで表同士が食い違うことがある。
導入後
受付表に1回入力してボタンを押すと、予定表と台帳に行が追加され、担当者宛てのメールが Gmail の下書き（送信前の状態）に作成されます。

お客様名などを変更して「転記する」を押してください。同じ内容のまま再度押しても、二重には登録されません。

予定表
台帳（番号は自動採番）
担当者宛てメール（下書き）
プログラムを見る（Apps Script）
/**
 * 受付シートの1行を、3か所へ自動で書き写す。
 *   ・予定表（日付と担当）
 *   ・台帳（通し番号をふって記録）
 *   ・担当者あての依頼メール（Gmail の下書き。送信は人が確認してから）
 *
 * 使い方：受付シートに1行入れて、C列の「状態」が空の行に対して
 *   メニュー「自動化 → 未処理の行を書き写す」を押す。
 *   すんだ行は「状態」に「済 2026/09/27 10:15」と入るので、二重に書き写されない。
 */

const SHEET_UKETSUKE = '受付';
const SHEET_YOTEI = '予定表';
const SHEET_DAICHO = '台帳';
const SHEET_TANTO = '担当者'; // A:名前 B:メールアドレス

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('自動化')
    .addItem('未処理の行を書き写す', 'tenkiAll')
    .addToUi();
}

function tenkiAll() {
  const ss = SpreadsheetApp.getActive();
  const src = ss.getSheetByName(SHEET_UKETSUKE);
  const values = src.getDataRange().getValues();
  const header = values[0];
  const col = name => {
    const i = header.indexOf(name);
    if (i < 0) throw new Error('受付シートに「' + name + '」の列がありません');
    return i;
  };
  const c = {
    date: col('日付'), customer: col('お客様'), from: col('出発地'),
    to: col('到着地'), tanto: col('担当'), memo: col('備考'), status: col('状態'),
  };
  const mails = loadTanto_(ss);
  const daicho = ss.getSheetByName(SHEET_DAICHO);
  let seq = Math.max(daicho.getLastRow() - 1, 0);
  let done = 0;

  for (let r = 1; r < values.length; r++) {
    const row = values[r];
    if (row[c.status] || !row[c.customer]) continue;

    ss.getSheetByName(SHEET_YOTEI).appendRow([row[c.date], row[c.tanto], row[c.customer], row[c.from] + ' → ' + row[c.to]]);

    seq += 1;
    const no = 'K-' + Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyMM') + '-' + String(seq).padStart(3, '0');
    daicho.appendRow([no, row[c.date], row[c.customer], row[c.from], row[c.to], row[c.tanto], row[c.memo]]);

    const to = mails[row[c.tanto]];
    if (to) {
      const d = Utilities.formatDate(new Date(row[c.date]), 'Asia/Tokyo', 'M月d日');
      GmailApp.createDraft(to, '【依頼】' + d + ' ' + row[c.customer] + '様',
        row[c.tanto] + 'さん\n\n次の件をお願いします。\n\n' +
        '番号：' + no + '\n日付：' + d + '\nお客様：' + row[c.customer] + '様\n' +
        '経路：' + row[c.from] + ' → ' + row[c.to] + '\n備考：' + (row[c.memo] || 'なし') + '\n');
    }

    src.getRange(r + 1, c.status + 1)
      .setValue('済 ' + Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm'));
    done += 1;
  }
  SpreadsheetApp.getActive().toast(done + '件を書き写しました（メールは下書きに入っています）');
}

function loadTanto_(ss) {
  const out = {};
  ss.getSheetByName(SHEET_TANTO).getDataRange().getValues().slice(1)
    .forEach(([name, mail]) => { if (name && mail) out[name] = mail; });
  return out;
}

請求書を1枚ずつ作成し、メールに添付している

現在
取引先ごとにひな形を開いて社名と金額を入力し、PDFで保存して、メールに添付している。
導入後
請求一覧でボタンを押すと、取引先ごとの請求書PDFが作成され、PDFを添付したメールが Gmail の下書きに並びます。送信はご確認のうえご自身で行うため、宛先や金額の誤りに送信前に気づけます。

表の金額や社名を変更して「請求書を作成」を押してください。メールアドレスが空欄の取引先は作成されません。

請求番号会社名金額メールアドレス

プログラムを見る（Apps Script）
/**
 * 請求一覧の1行ずつを、Google ドキュメントのひな形に差し込んで PDF にし、
 * その PDF を添付した Gmail の下書きを作る。送信はしない（人が見てから送る）。
 *
 * ひな形のドキュメントには {{会社名}} {{金額}} {{請求番号}} {{支払期限}} のように書いておく。
 * 一覧の見出しと {{ }} の中の名前が同じなら、列が増えてもそのまま差し込まれる。
 */

const TEMPLATE_DOC_ID = 'ここにひな形ドキュメントのID';
const OUTPUT_FOLDER_ID = 'ここにPDFを置くフォルダのID';

function makeInvoices() {
  const sheet = SpreadsheetApp.getActive().getSheetByName('請求一覧');
  const [header, ...rows] = sheet.getDataRange().getValues();
  const iMail = header.indexOf('メール');
  const iDone = header.indexOf('下書き');
  const folder = DriveApp.getFolderById(OUTPUT_FOLDER_ID);

  rows.forEach((row, k) => {
    if (row[iDone] || !row[iMail]) return;

    const copy = DriveApp.getFileById(TEMPLATE_DOC_ID).makeCopy('tmp_' + k, folder);
    const doc = DocumentApp.openById(copy.getId());
    const body = doc.getBody();
    header.forEach((name, j) => {
      const v = row[j] instanceof Date
        ? Utilities.formatDate(row[j], 'Asia/Tokyo', 'yyyy年M月d日')
        : typeof row[j] === 'number' ? row[j].toLocaleString('ja-JP') : String(row[j]);
      body.replaceText('\\{\\{' + name + '\\}\\}', v);
    });
    doc.saveAndClose();

    const fileName = '請求書_' + row[header.indexOf('請求番号')] + '_' + row[header.indexOf('会社名')] + '.pdf';
    const pdf = copy.getAs('application/pdf').setName(fileName);
    folder.createFile(pdf);
    copy.setTrashed(true); // 作業用のコピーはゴミ箱へ（30日は戻せる）

    GmailApp.createDraft(row[iMail], '請求書送付のご案内（' + row[header.indexOf('請求番号')] + '）',
      row[header.indexOf('会社名')] + '\nご担当者様\n\nいつもお世話になっております。\n' +
      '請求書を添付いたしましたので、ご確認をお願いいたします。\n',
      { attachments: [pdf] });

    sheet.getRange(k + 2, iDone + 1).setValue(new Date());
  });
}

問い合わせを1件ずつ読んで振り分けている

現在
届いた問い合わせをすべて読み、急ぎかどうかを判断して、返信を一から書いている。
導入後
問い合わせが届くと AI が内容を読み、「見積もり」「不具合」などに分類します。急ぎのものは担当者へすぐにメールで通知し、返信文の下書きもシートに入ります。

左の問い合わせを1つ選ぶと、右側に分類の結果と返信文の下書きが表示されます（表示内容は見本として用意したものです）。

分類の結果左の問い合わせを選ぶと表示されます
プログラムを見る（Apps Script）
/**
 * お問い合わせフォームの回答を、生成 AI で仕分けてシートに書き足す。
 *   種類（見積もり・不具合・解約・その他）／急ぎかどうか／返信の下書き
 *
 * フォームの送信をきっかけに動く（トリガー「フォーム送信時」に onFormSubmit を登録）。
 * API キーはコードに書かず、スクリプト プロパティ AI_API_KEY に依頼者ご自身で入れていただく。
 * ここでは Anthropic の API を使う例。Gemini や OpenAI でも、呼び出し部分を差し替えるだけで動く。
 */

const MODEL = 'claude-haiku-4-5-20251001'; // 仕分けは安くて速いモデルで足りる

function onFormSubmit(e) {
  const sheet = e.range.getSheet();
  const row = e.range.getRow();
  const text = e.namedValues['お問い合わせ内容'][0];

  const result = classify_(text);
  const header = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const put = (name, v) => {
    let c = header.indexOf(name) + 1;
    if (c === 0) { c = header.length + 1; sheet.getRange(1, c).setValue(name); header.push(name); }
    sheet.getRange(row, c).setValue(v);
  };
  put('種類', result.category);
  put('急ぎ', result.urgent ? '急ぎ' : '');
  put('返信の下書き', result.reply);

  // 急ぎのときだけ、社内の担当者へ知らせる（お客様へは送らない）
  const notifyTo = PropertiesService.getScriptProperties().getProperty('NOTIFY_TO');
  if (result.urgent && notifyTo) {
    MailApp.sendEmail(notifyTo, '【急ぎの問い合わせ】' + result.category, text);
  }
}

function classify_(text) {
  const key = PropertiesService.getScriptProperties().getProperty('AI_API_KEY');
  const prompt =
    '次のお問い合わせを読んで、JSON だけを返してください。\n' +
    '{"category": "見積もり|不具合|解約|その他", "urgent": true/false, "reply": "返信の下書き（200字以内・丁寧語）"}\n' +
    '急ぎとは、業務が止まっている・今日中の対応を求めている・強い不満がある場合です。\n\n' +
    'お問い合わせ：\n' + text;

  const res = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', {
    method: 'post',
    contentType: 'application/json',
    headers: { 'x-api-key': key, 'anthropic-version': '2023-06-01' },
    payload: JSON.stringify({ model: MODEL, max_tokens: 600, messages: [{ role: 'user', content: prompt }] }),
    muteHttpExceptions: true,
  });
  if (res.getResponseCode() !== 200) throw new Error('AI の呼び出しに失敗：' + res.getContentText());
  const out = JSON.parse(res.getContentText()).content[0].text;
  return JSON.parse(out.slice(out.indexOf('{'), out.lastIndexOf('}') + 1));
}

期限を記憶やカレンダーで管理している

現在
保険の更新や支払いの期限に、気づくのが遅れることがある。
導入後
毎朝、期限の一覧を自動で確認し、30日前・7日前・前日・当日・期限切れの項目を1通のメールでお知らせします。該当がない日はメールを送りません。

つまみを動かして日付を変えると、その日の朝に届くメールが表示されます。

項目期限対応
この日の朝に届くメール
プログラムを見る（Apps Script）
/**
 * 期限の近いものを、毎朝まとめて知らせる。
 *   契約の更新日・在庫の賞味期限・資格の有効期限・請求の支払期限など、列の名前を変えるだけで使い回せる。
 *
 * 「期限」の列が今日から DAYS_BEFORE 日以内で、「対応」が空の行を拾う。
 * 同じものを毎日しつこく知らせないように、30日前・7日前・前日・当日・過ぎたもの、の節目だけ知らせる。
 * トリガー：時間主導型 → 日付ベース → 午前8〜9時 に notifyDeadlines を登録。
 */

const SHEET_NAME = '期限一覧';
const NOTICE_DAYS = [30, 7, 1, 0];
const NOTIFY_TO = 'ここに知らせる先のメールアドレス';

function notifyDeadlines() {
  const [header, ...rows] = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME).getDataRange().getValues();
  const iName = header.indexOf('項目'), iDue = header.indexOf('期限'), iDone = header.indexOf('対応');
  const today = new Date(); today.setHours(0, 0, 0, 0);

  const lines = { over: [], hit: [] };
  rows.forEach(r => {
    if (r[iDone] || !(r[iDue] instanceof Date)) return;
    const due = new Date(r[iDue]); due.setHours(0, 0, 0, 0);
    const left = Math.round((due - today) / 86400000);
    const d = Utilities.formatDate(due, 'Asia/Tokyo', 'M/d');
    if (left < 0) lines.over.push('・' + r[iName] + '（' + d + '・' + (-left) + '日過ぎ）');
    else if (NOTICE_DAYS.includes(left)) lines.hit.push('・' + r[iName] + '（' + d + '・' + (left === 0 ? '今日' : 'あと' + left + '日') + '）');
  });
  if (!lines.over.length && !lines.hit.length) return; // 何もない日は送らない

  const body =
    (lines.over.length ? '■ 期限を過ぎたもの\n' + lines.over.join('\n') + '\n\n' : '') +
    (lines.hit.length ? '■ 期限が近いもの\n' + lines.hit.join('\n') + '\n\n' : '') +
    '対応したら「対応」の列に日付か「済」を入れてください。知らせが止まります。\n' +
    SpreadsheetApp.getActive().getUrl();
  MailApp.sendEmail(NOTIFY_TO, '期限のお知らせ（' + (lines.over.length + lines.hit.length) + '件）', body);
}

毎月、同じ集計表を作り直している

現在
毎月CSVを取り込み、数式をコピーしてグラフを作り直している。数式のコピー漏れで、数字がずれることもある。
導入後
CSVを所定のフォルダに置いてボタンを押すと、担当者別・商品別の合計とグラフが入った月別のシートが作成されます。

明細の金額や担当者を変更して「集計する」を押してください。

プログラムを見る（Apps Script）
/**
 * 売上の明細（毎月 CSV で落としてくるもの）から、月次の集計表とグラフを作る。
 *
 * 1. 「取込」シートに CSV を貼る（またはフォルダに置いた CSV を読み込む importLatestCsv）
 * 2. makeMonthlyReport を押すと、「月次_2026-09」のようなシートができ、
 *    担当者別・商品別の売上と、棒グラフが並ぶ
 * 毎月同じ形のシートができるので、作り直しの手間と、式のコピー漏れがなくなる。
 */

const CSV_FOLDER_ID = 'ここにCSVを置くフォルダのID';

function importLatestCsv() {
  const files = DriveApp.getFolderById(CSV_FOLDER_ID).getFilesByType(MimeType.CSV);
  let latest = null;
  while (files.hasNext()) {
    const f = files.next();
    if (!latest || f.getLastUpdated() > latest.getLastUpdated()) latest = f;
  }
  if (!latest) throw new Error('フォルダに CSV がありません');
  // 会計ソフトの CSV は Shift_JIS のことが多い
  const text = latest.getBlob().getDataAsString('Shift_JIS');
  const data = Utilities.parseCsv(text);
  const sheet = SpreadsheetApp.getActive().getSheetByName('取込');
  sheet.clearContents();
  sheet.getRange(1, 1, data.length, data[0].length).setValues(data);
}

function makeMonthlyReport() {
  const ss = SpreadsheetApp.getActive();
  const [header, ...rows] = ss.getSheetByName('取込').getDataRange().getValues();
  const iDate = header.indexOf('日付'), iWho = header.indexOf('担当'),
        iItem = header.indexOf('商品'), iAmt = header.indexOf('金額');

  const month = Utilities.formatDate(new Date(rows[0][iDate]), 'Asia/Tokyo', 'yyyy-MM');
  const sum = key => {
    const m = {};
    rows.forEach(r => { m[r[key]] = (m[r[key]] || 0) + Number(r[iAmt]); });
    return Object.entries(m).sort((a, b) => b[1] - a[1]);
  };

  const name = '月次_' + month;
  const old = ss.getSheetByName(name);
  if (old) ss.deleteSheet(old); // 同じ月を作り直すときは上書き
  const out = ss.insertSheet(name);

  const byWho = sum(iWho), byItem = sum(iItem);
  out.getRange(1, 1, 1, 2).setValues([['担当', '売上']]);
  out.getRange(2, 1, byWho.length, 2).setValues(byWho);
  out.getRange(1, 4, 1, 2).setValues([['商品', '売上']]);
  out.getRange(2, 4, byItem.length, 2).setValues(byItem);
  out.getRange('B:B').setNumberFormat('#,##0');
  out.getRange('E:E').setNumberFormat('#,##0');

  out.insertChart(out.newChart().asBarChart()
    .addRange(out.getRange(1, 1, byWho.length + 1, 2))
    .setPosition(2, 7, 0, 0).setOption('title', month + ' 担当者別の売上').build());
}
